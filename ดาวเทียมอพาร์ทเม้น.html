function getSummary() { const sheet = getSheet('Transactions'); const data = sheet.getDataRange().getValues(); const headers = data[0]; const typeCol = headers.indexOf('type'); const amountCol = headers.indexOf('amount'); const monthCol = headers.indexOf('month'); const incMonths = new Set(), expMonths = new Set(); let totalInc = 0, totalExp = 0; for (let i = 1; i < data.length; i++) { const type = String(data[i][typeCol] || ''); const amount = parseFloat(data[i][amountCol] || 0); const month = String(data[i][monthCol] || ''); if (!type || !amount) continue; if (type.endsWith('-out')) { totalExp += amount; if (month) expMonths.add(month); } else { totalInc += amount; if (month) incMonths.add(month); } } return { success: true, totalInc, totalExp, incMonths: incMonths.size, expMonths: expMonths.size, avgInc: incMonths.size > 0 ? Math.round(totalInc / incMonths.size) : 0, avgExp: expMonths.size > 0 ? Math.round(totalExp / expMonths.size) : 0 }; }
